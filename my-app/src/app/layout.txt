import { useEffect } from 'react';
import { Stack } from 'expo-router';
import { ThemeProvider } from '../providers/ThemeProvider';
import { AuthProvider, useAuth } from '../contexto/AuthContext';
import { registrarTokenPush } from '../servicios/notificaciones';

// Creamos un sub-componente interno para poder usar el hook useAuth()
function ManejadorNotificaciones() {
  const { token } = useAuth(); // Sacamos el token JWT (Bearer) de la sesión

  useEffect(() => {
    // Si el usuario está logueado, le pedimos permisos de push y mandamos al backend
    if (token) {
      registrarTokenPush(token);
    }
  }, [token]);

  return (
    <Stack screenOptions={{ headerShown: false }}>
      <Stack.Screen name="index" options={{ title: 'Ingresar' }} />
    </Stack>
  );
}

export default function RootLayout() {
  return (
    <ThemeProvider>
      <AuthProvider>
        <ManejadorNotificaciones />
      </AuthProvider>
    </ThemeProvider>
  );
}

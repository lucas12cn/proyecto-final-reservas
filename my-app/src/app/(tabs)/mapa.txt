import React, { useState, useEffect } from 'react';
import { View, Text, StyleSheet, ActivityIndicator, TouchableOpacity } from 'react-native';
import MapView, { Marker, Callout } from 'react-native-maps';
import { router } from 'expo-router';
import Ionicons from '@react-native-vector-icons/ionicons';
import { obtenerEspacios } from '../../servicios/reservas';
import { useTema } from '../../hooks/use-theme';

export default function MapaComplejosScreen() {
  const { paleta } = useTema();
  const [complejos, setComplejos] = useState<any[]>([]);
  const [cargando, setCargando] = useState(true);

  // Centro aproximado de Villaguay
  const regionInicial = {
    latitude: -31.8644,
    longitude: -59.0272,
    latitudeDelta: 0.04,
    longitudeDelta: 0.04,
  };

  useEffect(() => {
    cargarDatosMapa();
  }, []);

  const cargarDatosMapa = async () => {
    try {
      const listaEspacios = await obtenerEspacios();
      const complejosAgrupados: Record<string, any> = {};
      
      listaEspacios.forEach((espacio: any) => {
        if (espacio.coordenadas && espacio.coordenadas.latitud && espacio.coordenadas.longitud) {
          const nombreComplejo = espacio.complejo || 'Complejo Deportivo';
          
          if (!complejosAgrupados[nombreComplejo]) {
            complejosAgrupados[nombreComplejo] = {
              nombre: nombreComplejo,
              latitud: espacio.coordenadas.latitud,
              longitud: espacio.coordenadas.longitud,
              canchas: []
            };
          }
          complejosAgrupados[nombreComplejo].canchas.push(espacio.nombre);
        }
      });

      setComplejos(Object.values(complejosAgrupados));
    } catch (error) {
      console.error("Error cargando datos para el mapa:", error);
    } finally {
      setCargando(false);
    }
  };

  return (
    <View style={estilos.pantalla}>
      {/* Cabecera Flotante */}
      <View style={[estilos.cabeceraFlotante, { backgroundColor: paleta.surface }]}>
        <TouchableOpacity onPress={() => router.back()} style={estilos.botonVolver}>
          <Ionicons name="arrow-back" size={24} color={paleta.text} />
        </TouchableOpacity>
        <Text style={[estilos.tituloCabecera, { color: paleta.text }]}>Mapa de Complejos</Text>
        <View style={{ width: 40 }} /> 
      </View>

      {cargando ? (
        <View style={[estilos.centrado, { backgroundColor: paleta.background }]}>
          <ActivityIndicator size="large" color={paleta.accent} />
          <Text style={{ color: paleta.text, marginTop: 15 }}>Cargando mapa nativo...</Text>
        </View>
      ) : (
        <MapView 
          style={StyleSheet.absoluteFill} 
          initialRegion={regionInicial}
          showsUserLocation={true}
        >
          {complejos.map((complejo, index) => (
            <Marker
              key={index}
              coordinate={{
                latitude: complejo.latitud,
                longitude: complejo.longitud,
              }}
            >
              <Ionicons name="location-sharp" size={45} color={paleta.accent} />
              
              <Callout tooltip>
                <View style={[estilos.burbujaInfo, { backgroundColor: paleta.surface, borderColor: paleta.border }]}>
                  <Text style={[estilos.tituloBurbuja, { color: paleta.text }]}>{complejo.nombre}</Text>
                  <Text style={[estilos.textoBurbuja, { color: paleta.muted }]}>
                    {complejo.canchas.length} {complejo.canchas.length === 1 ? 'cancha disponible' : 'canchas disponibles'}
                  </Text>
                  <Text style={[estilos.textoCanchas, { color: paleta.text }]}>
                    {complejo.canchas.join(', ')}
                  </Text>
                </View>
              </Callout>
            </Marker>
          ))}
        </MapView>
      )}
    </View>
  );
}

const estilos = StyleSheet.create({
  pantalla: { flex: 1 },
  centrado: { flex: 1, justifyContent: 'center', alignItems: 'center', zIndex: 5 },
  
  cabeceraFlotante: {
    position: 'absolute',
    top: 50,
    left: 20,
    right: 20,
    flexDirection: 'row',
    alignItems: 'center',
    justifyContent: 'space-between',
    padding: 12,
    borderRadius: 12,
    zIndex: 10,
    elevation: 5,
    shadowColor: '#000',
    shadowOffset: { width: 0, height: 2 },
    shadowOpacity: 0.25,
    shadowRadius: 3.84,
  },
  botonVolver: { padding: 5 },
  tituloCabecera: { fontSize: 18, fontWeight: 'bold' },
  
  burbujaInfo: {
    padding: 15,
    borderRadius: 12,
    borderWidth: 1,
    minWidth: 200,
    alignItems: 'center'
  },
  tituloBurbuja: { fontSize: 16, fontWeight: 'bold', marginBottom: 5 },
  textoBurbuja: { fontSize: 13, marginBottom: 8 },
  textoCanchas: { fontSize: 12, textAlign: 'center' }
});

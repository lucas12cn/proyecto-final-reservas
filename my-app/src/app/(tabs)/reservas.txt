import { View, Text, StyleSheet, FlatList, TouchableOpacity, Alert, ActivityIndicator, Modal } from 'react-native';
import { SafeAreaView } from 'react-native-safe-area-context';
import { useState, useCallback, useEffect } from 'react';
import { useFocusEffect, router } from 'expo-router';
import Ionicons from '@react-native-vector-icons/ionicons';
import QRCode from 'react-native-qrcode-svg';
import * as Network from 'expo-network';
import * as SQLite from 'expo-sqlite';
import { useTema } from '../../hooks/use-theme';
import { obtenerMisReservas, cancelarReserva, obtenerEspacios } from '../../servicios/reservas';
import { useAuth } from '../../contexto/AuthContext';

export default function MisReservasScreen() {
  const { paleta } = useTema();
  const { usuario } = useAuth();
  
  const [filtro, setFiltro] = useState<'proximas' | 'pasadas'>('proximas');
  const [reservas, setReservas] = useState<any[]>([]);
  const [espacios, setEspacios] = useState<Record<string, any>>({});
  const [cargando, setCargando] = useState(true);
  const [modoOffline, setModoOffline] = useState(false);
  const [dbLista, setDbLista] = useState(false);

  const [modalQRVisible, setModalQRVisible] = useState(false);
  const [qrSeleccionado, setQrSeleccionado] = useState<string | null>(null);

  useEffect(() => {
    async function prepararBD() {
      try {
        const db = await SQLite.openDatabaseAsync('mis_reservas_v5.db');
        await db.execAsync(`
          CREATE TABLE IF NOT EXISTS cache_reservas (
            filtro TEXT PRIMARY KEY,
            reservas_json TEXT,
            espacios_json TEXT
          );
        `);
        setDbLista(true);
      } catch (error) {
        console.log("Advertencia: Error inicializando SQLite:", error);
        setDbLista(false);
      }
    }
    prepararBD();
  }, []);

  const cargarDatos = async () => {
    // Si no hay usuario, cortamos la ejecución y apagamos la carga
    if (!usuario) {
      setCargando(false);
      return;
    }
    
    setCargando(true);
    try {
      const estadoRed = await Network.getNetworkStateAsync();

      if (estadoRed.isConnected && estadoRed.isInternetReachable) {
        // MODO ONLINE: Traemos los datos de la API sí o sí
        setModoOffline(false);
        const [listaReservas, listaEspacios] = await Promise.all([
          obtenerMisReservas(filtro),
          obtenerEspacios()
        ]);

        const mapaEspacios: Record<string, any> = {};
        listaEspacios.forEach((e: any) => mapaEspacios[e.id] = e);
        
        setEspacios(mapaEspacios);
        setReservas(listaReservas);

        // SOLO intentamos guardar en SQLite si la BD no está trabada
        if (dbLista) {
          try {
             const db = await SQLite.openDatabaseAsync('mis_reservas_v5.db');
             await db.runAsync(
               'INSERT OR REPLACE INTO cache_reservas (filtro, reservas_json, espacios_json) VALUES (?, ?, ?)',
               [filtro, JSON.stringify(listaReservas), JSON.stringify(mapaEspacios)]
             );
          } catch (sqliteError) {
             console.log("Advertencia: No se pudo guardar la caché local", sqliteError);
          }
        }

      } else {
        // MODO OFFLINE
        setModoOffline(true);
        if (dbLista) {
          try {
             const db = await SQLite.openDatabaseAsync('mis_reservas_v5.db');
             const fila = await db.getFirstAsync<{ reservas_json: string, espacios_json: string }>(
               'SELECT reservas_json, espacios_json FROM cache_reservas WHERE filtro = ?',
               [filtro]
             );

             if (fila) {
               setReservas(JSON.parse(fila.reservas_json));
               setEspacios(JSON.parse(fila.espacios_json));
             } else {
               setReservas([]);
               setEspacios({});
             }
          } catch (sqliteError) {
               console.log("Advertencia: No se pudo leer la caché local", sqliteError);
               setReservas([]);
               setEspacios({});
          }
        } else {
           setReservas([]);
           setEspacios({});
        }
      }
    } catch (error) {
      console.log("Error principal en cargarDatos:", error);
    } finally {
      // ESTO ES CLAVE: Pase lo que pase (haya fallado SQLite o no), apagamos la ruedita de carga
      setCargando(false);
    }
  };

  useFocusEffect(
    useCallback(() => {
      cargarDatos();
    }, [filtro, usuario, dbLista])
  );

  const confirmarCancelacion = (id: string) => {
    Alert.alert(
      'Cancelar Reserva',
      '¿Estás seguro de que querés cancelar este turno? Esta acción liberará la cancha.',
      [
        { text: 'No, mantener', style: 'cancel' },
        { 
          text: 'Sí, cancelar', 
          style: 'destructive',
          onPress: async () => {
            try {
              await cancelarReserva(id);
              Alert.alert('Éxito', 'La reserva fue cancelada.');
              cargarDatos(); 
            } catch (error: any) {
              Alert.alert('Error', error.message);
            }
          }
        }
      ]
    );
  };

  const formatearTurno = (turnoId: string) => {
    const partes = turnoId.split('-');
    if (partes.length < 5) return { fecha: '', hora: '' };
    const horaStr = partes[4] + ':00';
    const f = partes[3];
    const fechaStr = `${f.substring(0,4)}-${f.substring(4,6)}-${f.substring(6,8)}`;
    return { fecha: fechaStr, hora: horaStr };
  };

  const renderReserva = ({ item }: { item: any }) => {
    const espacioInfo = espacios[item.espacioId];
    const { fecha, hora } = formatearTurno(item.turnoId);
    
    let colorEstado = paleta.border;
    if (item.estado === 'confirmada') colorEstado = '#10b981'; 
    if (item.estado === 'cancelada') colorEstado = '#ef4444'; 
    if (item.estado === 'ausente') colorEstado = '#f59e0b'; 

    return (
      <View style={[estilos.tarjeta, { backgroundColor: paleta.surface, borderColor: paleta.border }]}>
        <View style={[estilos.cabeceraTarjeta, { borderBottomColor: paleta.border }]}>
          <View style={estilos.fechaContenedor}>
            <Ionicons name="calendar" size={18} color={paleta.accent} />
            <Text style={[estilos.textoFecha, { color: paleta.text }]}>{fecha}</Text>
            <Text style={[estilos.textoHora, { color: paleta.text }]}>{hora}</Text>
          </View>
          <View style={[estilos.etiquetaEstado, { backgroundColor: colorEstado }]}>
            <Text style={estilos.textoEstado}>
              {item.estado.charAt(0).toUpperCase() + item.estado.slice(1)}
            </Text>
          </View>
        </View>

        <View style={estilos.cuerpoTarjeta}>
          <Text style={[estilos.nombreCancha, { color: paleta.text }]}>
            {espacioInfo ? espacioInfo.nombre : 'Cargando...'}
          </Text>
          <Text style={{ color: paleta.muted }}>
            {espacioInfo ? espacioInfo.complejo : 'Espacio deportivo'}
          </Text>
          
          <View style={estilos.pieTarjeta}>
            <Text style={[estilos.precio, { color: paleta.text }]}>
              Total: ${espacioInfo ? espacioInfo.precioPorHora : '---'}
            </Text>
            
            <View style={estilos.accionesFila}>
              {item.estado === 'confirmada' && filtro === 'proximas' && !modoOffline && (
                <TouchableOpacity 
                  style={[estilos.botonAccion, { borderColor: '#ef4444' }]}
                  onPress={() => confirmarCancelacion(item.id)}
                >
                  <Ionicons name="close-circle-outline" size={16} color="#ef4444" />
                  <Text style={[estilos.textoBotonAccion, { color: '#ef4444' }]}>Cancelar</Text>
                </TouchableOpacity>
              )}

              {(item.estado === 'confirmada' || item.estado === 'cumplida') && (
                <TouchableOpacity 
                  style={[estilos.botonAccion, { borderColor: paleta.accent, backgroundColor: paleta.accent }]}
                  onPress={() => {
                    setQrSeleccionado(item.codigoQr || item.id);
                    setModalQRVisible(true);
                  }}
                >
                  <Ionicons name="qr-code-outline" size={16} color="white" />
                  <Text style={[estilos.textoBotonAccion, { color: 'white' }]}>QR</Text>
                </TouchableOpacity>
              )}
            </View>
          </View>
        </View>
      </View>
    );
  };

  return (
    <SafeAreaView style={[estilos.pantalla, { backgroundColor: paleta.background }]} edges={['top']}>
      <View style={estilos.cabecera}>
        <Text style={[estilos.tituloCentral, { color: paleta.text }]}>Mis Reservas</Text>
      </View>

      {!usuario ? (
        <View style={estilos.contenedorInvitado}>
          <Ionicons name="calendar-outline" size={100} color={paleta.muted} style={{ marginBottom: 15 }} />
          <Text style={[estilos.nombreInvitado, { color: paleta.text }]}>No estás logueado</Text>
          <Text style={[estilos.textoInvitado, { color: paleta.muted }]}>
            Iniciá sesión para poder gestionar y ver tu historial de reservas.
          </Text>
          <TouchableOpacity 
            style={[estilos.botonLogin, { backgroundColor: paleta.accent }]} 
            onPress={() => router.push('/login')}
          >
            <Text style={estilos.textoBotonLogin}>Iniciar Sesión</Text>
          </TouchableOpacity>
        </View>
      ) : (
        <>
          <View style={estilos.selectorFiltro}>
            <TouchableOpacity 
              style={[estilos.botonFiltro, filtro === 'proximas' && { borderBottomColor: paleta.accent, borderBottomWidth: 2 }]}
              onPress={() => setFiltro('proximas')}
            >
              <Text style={[estilos.textoFiltro, { color: filtro === 'proximas' ? paleta.accent : paleta.muted }]}>Próximas</Text>
            </TouchableOpacity>
            <TouchableOpacity 
              style={[estilos.botonFiltro, filtro === 'pasadas' && { borderBottomColor: paleta.accent, borderBottomWidth: 2 }]}
              onPress={() => setFiltro('pasadas')}
            >
              <Text style={[estilos.textoFiltro, { color: filtro === 'pasadas' ? paleta.accent : paleta.muted }]}>Historial</Text>
            </TouchableOpacity>
          </View>

          {modoOffline && (
            <View style={estilos.cartelOffline}>
              <Text style={estilos.textoOffline}>⚠️ Sin conexión. Mostrando reservas guardadas.</Text>
            </View>
          )}

          {cargando ? (
            <View style={estilos.vacioContenedor}>
              <ActivityIndicator size="large" color={paleta.accent} />
            </View>
          ) : (
            <FlatList
              data={reservas}
              keyExtractor={(item) => item.id}
              renderItem={renderReserva}
              contentContainerStyle={estilos.lista}
              showsVerticalScrollIndicator={false}
              ListEmptyComponent={
                <View style={estilos.vacioContenedor}>
                  <Ionicons name="calendar-clear-outline" size={50} color={paleta.muted} />
                  <Text style={[estilos.textoVacio, { color: paleta.muted }]}>
                    No tenés reservas {filtro === 'proximas' ? 'activas' : 'en tu historial'}.
                  </Text>
                </View>
              }
            />
          )}

          <Modal
            visible={modalQRVisible}
            transparent={true}
            animationType="fade"
            onRequestClose={() => setModalQRVisible(false)}
          >
            <View style={estilos.modalFondo}>
              <View style={[estilos.modalContenedor, { backgroundColor: paleta.background }]}>
                <Text style={[estilos.modalTitulo, { color: paleta.text }]}>Código de Ingreso</Text>
                
                <View style={[estilos.cajaQR, { backgroundColor: 'white' }]}>
                  {qrSeleccionado ? (
                    <QRCode value={qrSeleccionado} size={200} />
                  ) : (
                    <ActivityIndicator color="black" />
                  )}
                </View>

                <Text style={[estilos.modalTexto, { color: paleta.muted }]}>
                  Presentá este código en la entrada del complejo deportivo.
                </Text>

                <TouchableOpacity 
                  style={[estilos.botonCerrarModal, { backgroundColor: paleta.accent }]}
                  onPress={() => setModalQRVisible(false)}
                >
                  <Text style={estilos.textoBotonCerrar}>Cerrar</Text>
                </TouchableOpacity>
              </View>
            </View>
          </Modal>
        </>
      )}
    </SafeAreaView>
  );
}

const estilos = StyleSheet.create({
  pantalla: { flex: 1 },
  cabecera: { paddingHorizontal: 20, paddingTop: 10, paddingBottom: 15 },
  tituloCentral: { fontSize: 26, fontWeight: 'bold' },
  selectorFiltro: { flexDirection: 'row', paddingHorizontal: 20, marginBottom: 10 },
  botonFiltro: { flex: 1, paddingVertical: 10, alignItems: 'center' },
  textoFiltro: { fontSize: 16, fontWeight: 'bold' },
  cartelOffline: { backgroundColor: '#f59e0b', padding: 10, alignItems: 'center', marginHorizontal: 20, borderRadius: 8, marginBottom: 10 },
  textoOffline: { color: '#fff', fontWeight: 'bold', fontSize: 14 },
  lista: { padding: 20, paddingTop: 0, gap: 15 },
  tarjeta: { borderWidth: 1, borderRadius: 12, overflow: 'hidden' },
  cabeceraTarjeta: { flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center', padding: 15, borderBottomWidth: 1 },
  fechaContenedor: { flexDirection: 'row', alignItems: 'center', gap: 6 },
  textoFecha: { fontSize: 15, fontWeight: '600' },
  textoHora: { fontSize: 15, fontWeight: 'bold' },
  etiquetaEstado: { paddingHorizontal: 10, paddingVertical: 4, borderRadius: 20 },
  textoEstado: { color: 'white', fontSize: 12, fontWeight: 'bold' },
  cuerpoTarjeta: { padding: 15 },
  nombreCancha: { fontSize: 18, fontWeight: 'bold', marginBottom: 2 },
  pieTarjeta: { flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center', marginTop: 15 },
  precio: { fontSize: 16, fontWeight: '600' },
  accionesFila: { flexDirection: 'row', gap: 10 },
  botonAccion: { flexDirection: 'row', alignItems: 'center', gap: 5, paddingHorizontal: 12, paddingVertical: 6, borderRadius: 8, borderWidth: 1 },
  textoBotonAccion: { fontWeight: 'bold', fontSize: 14 },
  vacioContenedor: { alignItems: 'center', marginTop: 50, gap: 10 },
  textoVacio: { fontSize: 16, textAlign: 'center' },
  
  contenedorInvitado: { flex: 1, justifyContent: 'center', alignItems: 'center', paddingHorizontal: 20 },
  nombreInvitado: { fontSize: 24, fontWeight: 'bold', marginBottom: 10 },
  textoInvitado: { fontSize: 16, textAlign: 'center', marginBottom: 30, paddingHorizontal: 10 },
  botonLogin: { width: '100%', padding: 16, borderRadius: 12, alignItems: 'center' },
  textoBotonLogin: { color: 'white', fontSize: 18, fontWeight: 'bold' },

  modalFondo: { flex: 1, backgroundColor: 'rgba(0,0,0,0.6)', justifyContent: 'center', alignItems: 'center' },
  modalContenedor: { width: '85%', borderRadius: 20, padding: 25, alignItems: 'center' },
  modalTitulo: { fontSize: 22, fontWeight: 'bold', marginBottom: 20 },
  cajaQR: { padding: 20, borderRadius: 16, alignItems: 'center', justifyContent: 'center', elevation: 2, marginBottom: 20 },
  modalTexto: { textAlign: 'center', marginBottom: 25, fontSize: 15, lineHeight: 22 },
  botonCerrarModal: { width: '100%', paddingVertical: 14, borderRadius: 12, alignItems: 'center' },
  textoBotonCerrar: { color: 'white', fontSize: 16, fontWeight: 'bold' }
});
